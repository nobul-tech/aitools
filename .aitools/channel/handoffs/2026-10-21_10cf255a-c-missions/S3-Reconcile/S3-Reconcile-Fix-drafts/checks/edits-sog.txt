# --- Later lines of a 2-3 line command (#209) ---
# The first-token checks below see only line 1 (and an allowlisted first token,
# e.g. git, exits before them), so a second line could run rm with a glob, or
# sed/awk/cat on a file, unchecked. Each later line gets the same first-token
# checks. Observe mode logs and continues (it must not exit 0 before line 1 is
# checked); enforce blocks through violation().
later_violation() {  # message check-name
    if [ "$MODE_LATER" = "enforce" ]; then
        violation "$1" "enforce" "$2"
    fi
    emit_hook_event "hook_warn" "{\"check\":\"$2\",\"mode\":\"observe\",\"line\":\"later\"}"
    printf '%s [WOULD-BLOCK] (later line) %s | cmd: %s\n' \
        "$(date -u +%Y-%m-%dT%H:%M:%SZ)" "$1" "$COMMAND_FULL" >> "$LOG_FILE"
}
check_later_line() {  # line
    local line="$1" tok
    read -r tok _ <<< "$line"
    case "$tok" in
        rm)
            case "$line" in
                *'*'*|*'?'*) later_violation "USO: Simple Bash commands only --: Don't use glob patterns (*, ?) with 'rm' (line 2 or later of a multi-line command). Write a cleanup script listing the specific files." "glob-rm" ;;
            esac ;;
        cat|head|tail|sed|awk)
            case "$line" in
                *\|*) ;;  # pipeline: legitimate shell use, as on line 1
                *) later_violation "USO: Dedicated tools --: '$tok' outside a pipeline on line 2 or later of a multi-line command. Use the Read/Edit tool, or put the commands in a scratch script." "$tok" ;;
            esac ;;
        echo|printf)
            case "$line" in
                *\>*) later_violation "USO: Dedicated tools --: Use the Write tool instead of echo/printf redirection to create files (line 2 or later of a multi-line command)." "echo-redirect" ;;
            esac ;;
    esac
}
if [ "$NEWLINE_COUNT" -gt 0 ]; then
    _later="${COMMAND_FULL//\\n/$NL}"
    case "$_later" in
        *"$NL"*) _later="${_later#*"$NL"}" ;;
        *) _later="" ;;
    esac
    if [ -n "$_later" ]; then
        while IFS= read -r _line; do
            check_later_line "$_line"
        done <<< "$_later"
    fi
fi
