@@@FILE scripts/aitools-lib.sh SP-1 merge preview without a pipe (prompt_diff_review)
@@@OLD
        printf '%s\n' "$merged" | head -30 | while IFS= read -r line; do
            printf '  | %s\n' "$line" > /dev/tty
        done
@@@NEW
        # A here-string, not `printf | head`: under pipefail a merged file larger
        # than the pipe buffer made printf die of SIGPIPE (exit 141, set -e) (#184).
        local _pv_n=0 _pv_line
        while IFS= read -r _pv_line; do
            _pv_n=$((_pv_n + 1))
            if [ "$_pv_n" -gt 30 ]; then break; fi
            printf '  | %s\n' "$_pv_line" > /dev/tty
        done <<< "$merged"
@@@END
