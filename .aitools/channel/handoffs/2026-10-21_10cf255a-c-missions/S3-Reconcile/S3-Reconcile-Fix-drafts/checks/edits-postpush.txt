Drafts for #185: check-post-push steps 30 and 31 extract return values and
tracker cases with `\w[\w-]*` / `[\w-]+`, so the value "accept & adopt" is never
seen (bash and PS1 both miss it, so step 31's comparison passes vacuously).
Both twins. S3-Reconcile-Fix-Checks, 2026-10-26. Base: origin/main 60e7fd7.
Test: .scratch/session-rcfixk/test-postpush-regex.sh (runs every extraction,
old and new, on origin/main's libs).

@@@FILE scripts/check-post-push.sh PP-1 step 30: values with spaces, line-wise loop
@@@OLD
    sh_returns=$(perl -ne 'print "$1\n" if /MANAGED_FILE_RESULT="(\w[\w-]*)"/' "$sh_lib" | sort -u | paste -s -d , -)
@@@NEW
    # [^"]+ so "accept & adopt" is seen (#185); one value per line for the loop.
    sh_returns_lines=$(perl -ne 'print "$1\n" if /MANAGED_FILE_RESULT="([^"]+)"/' "$sh_lib" | sort -u)
    sh_returns=$(printf '%s\n' "$sh_returns_lines" | paste -s -d , -)
@@@END

@@@FILE scripts/check-post-push.sh PP-2 step 30: no word splitting of values
@@@OLD
        for val in $(echo "$sh_returns" | tr ',' ' '); do
            if ! grep -q "$val" "$caller"; then
                missing="${missing}${missing:+; }$cname missing $val"
            fi
        done
@@@NEW
        while IFS= read -r val; do
            [ -n "$val" ] || continue
            if ! grep -qF -- "$val" "$caller"; then
                missing="${missing}${missing:+; }$cname missing $val"
            fi
        done <<< "$sh_returns_lines"
@@@END
