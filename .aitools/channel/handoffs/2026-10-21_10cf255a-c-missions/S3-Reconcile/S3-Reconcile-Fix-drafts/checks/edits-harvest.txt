def _scratch_dir_recent(session_dir: Path, now: float) -> bool:
    """True if the dir or a top-level file in it changed within
    LIVE_MTIME_SECONDS (a session or delegate may still be writing it).
    Conservative: True when it cannot be read."""
    try:
        newest = session_dir.stat().st_mtime
        for p in session_dir.iterdir():
            if p.is_file():
                newest = max(newest, p.stat().st_mtime)
    except OSError as e:
        log_detail(f"cannot stat {session_dir.name}, treating as live: {e}")
        return True
    return (now - newest) < LIVE_MTIME_SECONDS
