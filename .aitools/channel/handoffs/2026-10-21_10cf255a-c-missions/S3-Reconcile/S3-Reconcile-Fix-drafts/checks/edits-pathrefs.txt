@@@OLD
def exists(root: Path, cand: str) -> bool:
    if cand.endswith(".sh/.ps1"):
        stem = cand[: -len(".sh/.ps1")]
        return (root / f"{stem}.sh").is_file() and (root / f"{stem}.ps1").is_file()
    target = root / cand
@@@NEW
PAIR = re.compile(r"^(.+)\.([A-Za-z0-9]+)/\.([A-Za-z0-9]+)$")


def exists(root: Path, cand: str) -> bool:
    pair = PAIR.match(cand)
    if pair:
        stem, ext1, ext2 = pair.groups()
        return (root / f"{stem}.{ext1}").is_file() and (root / f"{stem}.{ext2}").is_file()
    target = root / cand
@@@END
