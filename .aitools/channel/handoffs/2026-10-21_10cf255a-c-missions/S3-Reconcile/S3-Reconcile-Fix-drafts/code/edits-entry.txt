# build-deploy exits 0 when it skips the build (no node, no dotprofile, resolver

# not OK "Rebuilt". Parity: aitools.ps1 Step 2.

printf '%s\n' "$build_out" >> "$LOG_FILE"
if [ "$build_rc" -ne 0 ]; then
    log_error "build failed (see $LOG_DISPLAY)"
    return 1
fi
# grep exits 1 when no line matches (the normal case: the build ran); the empty
# result is the check.
build_skip=$(grep -F -m 1 "skipping deploy build" <<< "$build_out") || build_skip=""
if [ -n "$build_skip" ]; then
    build_why=$(perl -pe 's/\e\[[0-9;]*m//g; s/^.*?\[warn\] //' <<< "$build_skip") || build_why="$build_skip"
    log_warn "Deploy build skipped: $build_why"
    write_summary WARN "deploy build" "skipped (see deploy log)"
else
    log_ok "Rebuilt"
fi
@@@END
