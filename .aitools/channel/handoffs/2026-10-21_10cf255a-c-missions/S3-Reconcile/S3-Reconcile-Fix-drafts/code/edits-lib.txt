# ---------------------------------------------------------------------------
# Find-GitBash -- the harness-managed bash on Windows: Git for Windows' bash.exe
# (.claude/rules/cross-platform.md "Windows: managed bash"). Never PATH or
# Get-Command, which often resolve to the WSL shim. Returns the first path that
# exists, or $null. One list for every caller (setup-bash.ps1, the aitools.ps1
# build step; check-pre-push.ps1 can switch to it).
# KnownPaths (script-standards.md "KnownPaths empirical verification"): each path
# is UNVERIFIED until it is recorded on a Windows machine; then replace its
# marker with "# Verified: YYYY-MM-DD (vX.Y.Z)".
# ---------------------------------------------------------------------------
function Find-GitBash {
    $candidates = @()
    if ($env:ProgramFiles) { $candidates += (Join-Path $env:ProgramFiles "Git\bin\bash.exe") }                # UNVERIFIED
    if (${env:ProgramFiles(x86)}) { $candidates += (Join-Path ${env:ProgramFiles(x86)} "Git\bin\bash.exe") }  # UNVERIFIED
    if ($env:LOCALAPPDATA) { $candidates += (Join-Path $env:LOCALAPPDATA "Programs\Git\bin\bash.exe") }       # UNVERIFIED
    foreach ($p in $candidates) {
        if (Test-Path -LiteralPath $p) { return $p }
    }
    return $null
}
