@@@NEW
        _heredoc_marker="__SKILL_$(echo "$_sname" | tr '[:lower:]-' '[:upper:]_')__"
        # A line equal to the marker would end the heredoc early (embed_file_bash
        # checks the same). One perl, no pipe: grep -q could SIGPIPE it.
        if M="$_heredoc_marker" perl -0777 -ne 's/\r\n/\n/g; exit(/^\Q$ENV{M}\E$/m ? 0 : 1)' "$_sfile"; then
            blog_error "Cannot embed $_sfile: it contains the heredoc delimiter line $_heredoc_marker"
            exit 1
        fi
        echo "cat > \"\$_skill_tmp/${_sname}.md\" <<'${_heredoc_marker}'"
        # CRLF -> LF and one final newline, so the marker always starts its own line
        perl -0777 -pe 's/\r\n/\n/g; s/\n?\z/\n/' "$_sfile"
        echo "${_heredoc_marker}"
@@@END
